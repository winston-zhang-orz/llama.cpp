## Phase A — SET_ROWS gate tightening

By the end of Phase A, `supports_op` for `SET_ROWS` returns true only for shape patterns we already pass on `test-backend-ops`. This makes our existing post-MVP-1 cast logic provably safe and forces real-model KV-cache writes back to CPU through ggml's scheduler.

### Task A1: Tighten the `SET_ROWS` gate

**Files:**
- Modify: `ggml/src/ggml-gcu/ggml-gcu.cpp` (the `supports_op` switch case for `GGML_OP_SET_ROWS`)

- [ ] **Step 1: Replace the current `SET_ROWS` case in `ggml_backend_gcu_device_supports_op`** with the tightened version:

```cpp
case GGML_OP_SET_ROWS: {
    const ggml_tensor * src = op->src[0];
    const ggml_tensor * idx = op->src[1];
    const ggml_tensor * dst = op->src[2];
    if (!src || !idx || !dst) return false;
    if (idx->type != GGML_TYPE_I32 && idx->type != GGML_TYPE_I64) return false;
    if (!gcu_dtype_supported(src->type) || !gcu_dtype_supported(dst->type)) return false;
    // Allow F32->F16 cast (cache-style writes) AND same-dtype.
    if (src->type != dst->type &&
        !(src->type == GGML_TYPE_F32 && dst->type == GGML_TYPE_F16)) return false;
    if (!ggml_is_contiguous(src) || !ggml_is_contiguous(dst)) return false;
    if (src->ne[0] != dst->ne[0]) return false;
    // MVP-2 explicit decision: 4D KV cache shapes go to CPU.
    // We only accept what test-backend-ops exercises and produces OK on
    // the SDK we have: ne[2] == 1 && ne[3] == 1 for both src and dst.
    if (src->ne[2] != 1 || src->ne[3] != 1) return false;
    if (dst->ne[2] != 1 || dst->ne[3] != 1) return false;
    if (idx->ne[1] != 1 || idx->ne[2] != 1 || idx->ne[3] != 1) return false;
    if (!ggml_is_contiguous(idx)) return false;
    if (src->ne[1] * src->ne[2] * src->ne[3] != idx->ne[0]) return false;
    return true;
}
```

```cpp
static bool gcu_op_cpy(ggml_backend_gcu_context * ctx, ggml_tensor * dst) {
    ggml_tensor * src = dst->src[0];

    // Same dtype + contiguous: fast path (existing).
    if (src->type == dst->type && ggml_is_contiguous(src) && ggml_is_contiguous(dst)) {
        GGML_ASSERT(ggml_nbytes(src) == ggml_nbytes(dst));
        TOPS_CHECK(topsMemcpyAsync(dst->data, src->data, ggml_nbytes(src),
                                   topsMemcpyDeviceToDevice, ctx->stream));
        return true;
    }

    // Dtype conversion path: src and dst share shape, differ in dtype.
    GGML_ASSERT(ggml_is_contiguous(src) && ggml_is_contiguous(dst));
    GGML_ASSERT(ggml_nelements(src) == ggml_nelements(dst));

    gcu_tensor_dims dout, dlhs;

    TOPSATEN_CHECK(topsatenTo(out, in, target,
```

- [ ] **Step 1: Add the op handler**:

```cpp
static bool gcu_op_soft_max(ggml_backend_gcu_context * ctx, ggml_tensor * dst) {
    ggml_tensor * x    = dst->src[0];
    ggml_tensor * mask = dst->src[1];   // may be nullptr
    float scale, max_bias;
    memcpy(&scale,    (const float *)dst->op_params + 0, sizeof(float));
    memcpy(&max_bias, (const float *)dst->op_params + 1, sizeof(float));
    GGML_ASSERT(max_bias == 0.0f);

    // Allocate scratch for scaled (and mask-added) input. Same dtype/shape as dst.
    const size_t bytes = ggml_nbytes(dst);
    void * scratch = ctx->pool.alloc(bytes);

    gcu_tensor_dims d_dst, d_x, d_mask;
    topsatenTensor out_tt = make_topsaten_tensor(dst, d_dst);
    topsatenTensor x_tt   = make_topsaten_tensor(x,   d_x);

    // Build a tensor view of `scratch` matching dst's shape/dtype.
    topsatenTensor scratch_tt = make_topsaten_tensor(dst, d_dst);
    // Override the data pointer to point at scratch (make_topsaten_tensor does
    // not let us do this directly; rebuild manually):
    {
        const size_t bpe = ggml_type_size(dst->type);
        int rank = ggml_n_dims(dst); if (rank < 1) rank = 1;
        for (int i = 0; i < rank; i++) {
            d_dst.dims[i] = dst->ne[rank - 1 - i];
            d_dst.strs[i] = dst->nb[rank - 1 - i] / (int64_t) bpe;
        }
        scratch_tt = topsatenTensor(topsatenSize_t(d_dst.dims, rank),
                                    topsatenSize_t(d_dst.strs, rank),
                                    ggml_to_topsaten_dtype(dst->type), scratch);
    }

    // scratch = x * scale  (using existing scalar-mul path).
    {
        topsatenScalar_t s; s.dtype = TOPSATEN_DATA_FP32; s.fval = scale;
        TOPSATEN_CHECK(topsatenMul(scratch_tt, x_tt, s, ctx->stream));
    }

    // If mask present, scratch = scratch + mask  (alpha=1).
    if (mask) {
        topsatenTensor mask_tt = make_topsaten_tensor(mask, d_mask);
        topsatenScalar_t alpha; alpha.dtype = TOPSATEN_DATA_FP32; alpha.fval = 1.0;
        TOPSATEN_CHECK(topsatenAdd(scratch_tt, scratch_tt, mask_tt, alpha, ctx->stream));
    }

    // out = softmax(scratch, dim=-1).
    // PyTorch order: dim=-1 is the last dim. For our PyTorch-order conversion,
    // that corresponds to ggml's ne[0] (the innermost / fastest dim), which
    // becomes the last PyTorch dim. So pass dims = rank - 1 = -1 effectively.
    int rank = ggml_n_dims(dst); if (rank < 1) rank = 1;
    TOPSATEN_CHECK(topsatenSoftmaxForward(out_tt, scratch_tt, rank - 1, ctx->stream));

    TOPS_CHECK(topsStreamSynchronize(ctx->stream));
    ctx->pool.free(scratch, bytes);
    return true;
}
```

```cpp
// Returns a device pointer to a F32 ones buffer of at least `count` elements.
// Grows on demand. Always F32; callers cast as needed.
static void * gcu_get_ones_f32(ggml_backend_gcu_context * ctx, int64_t count) {
    if (ctx->ones_n0_count >= count) return ctx->ones_n0;
    if (ctx->ones_n0) {
        ctx->pool.free(ctx->ones_n0, ctx->ones_n0_bytes);
    }
    const size_t bytes = (size_t) count * sizeof(float);
    ctx->ones_n0       = ctx->pool.alloc(bytes);
    ctx->ones_n0_bytes = bytes;
    ctx->ones_n0_count = count;

    // Fill with 1.0f. Use topsatenAdd of zero+scalar for portability:
    // 1) memset to zero
    TOPS_CHECK(topsMemsetAsync(ctx->ones_n0, 0, bytes, ctx->stream));
    // 2) add scalar 1.0f
    int64_t dims[1] = { count };
    int64_t strs[1] = { 1 };
    topsatenTensor t(topsatenSize_t(dims, 1), topsatenSize_t(strs, 1),
                     TOPSATEN_DATA_FP32, ctx->ones_n0);
    topsatenScalar_t one_lhs; one_lhs.dtype = TOPSATEN_DATA_FP32; one_lhs.fval = 1.0;
    topsatenScalar_t alpha;   alpha.dtype   = TOPSATEN_DATA_FP32; alpha.fval   = 1.0;
    // t = 1.0 + 1.0 * t (where t is currently 0) → t becomes 1.0
    // topsatenAdd has a (scalar, tensor, alpha) overload (see Phase E0 in MVP-1 plan).
    TOPSATEN_CHECK(topsatenAdd(t, one_lhs, t, alpha, ctx->stream));
    return ctx->ones_n0;
}
```

(Note: if a `topsatenFull(out, scalar)` op exists in `topsaten_ops.h`, use that instead — cleaner. Check during this task.)

```cpp
static bool gcu_op_rms_norm(ggml_backend_gcu_context * ctx, ggml_tensor * dst) {
    ggml_tensor * src = dst->src[0];
    float eps;
    memcpy(&eps, dst->op_params, sizeof(float));

    const int64_t hidden_size = src->ne[0];

    // Get a F32 ones buffer of size hidden_size. Cast to dst's dtype on the fly
    // by wrapping it as F32 and letting topsvllmRmsNorm handle it; if that
    // fails, fall back to allocating a per-call dst-typed ones via topsatenTo.
    void * ones_dev = gcu_get_ones_f32(ctx, hidden_size);

    int64_t gamma_d[1] = { hidden_size };
    int64_t gamma_s[1] = { 1 };
    topsatenTensor gamma_t(topsatenSize_t(gamma_d, 1), topsatenSize_t(gamma_s, 1),
```

```cpp
// Builds an interleaved [max_pos, n_dims] cos/sin table per topsvllm
// convention: row p contains cos values for theta_i*p in the first n_dims/2
// slots followed by sin values. Final F32 buffer.
//
// theta_i = freq_base^(-2i / n_dims) * freq_scale
//
// max_pos = max(positions[]) + 1; we conservatively use n_tokens + n_ctx_orig
// to cover any sliding-window pattern.
static void gcu_build_rope_cos_sin_host(int n_dims, int max_pos,
                                        float freq_base, float freq_scale,
                                        std::vector<float> & out) {
    const int half = n_dims / 2;
    out.resize((size_t) max_pos * (size_t) n_dims);
    for (int p = 0; p < max_pos; p++) {
        float * row = out.data() + (size_t) p * n_dims;
        for (int i = 0; i < half; i++) {
            const float theta = std::pow(freq_base, -2.0f * i / n_dims) * freq_scale;
            const float angle = (float) p * theta;
            row[i]        = std::cos(angle);
            row[i + half] = std::sin(angle);
        }
    }
}
```

```cpp
static bool gcu_op_rope(ggml_backend_gcu_context * ctx, ggml_tensor * dst) {
    ggml_tensor * x   = dst->src[0];
    ggml_tensor * pos = dst->src[1];                        // I32, length n_tokens
    // src[2] is optional ext-factor freqs (not used in mode 0).

    const int32_t n_dims      = ((const int32_t *)dst->op_params)[1];
    const int32_t mode        = ((const int32_t *)dst->op_params)[2];
    const int32_t n_ctx_orig  = ((const int32_t *)dst->op_params)[4];
    const float   freq_base   = *(const float *)((const int32_t *)dst->op_params + 5);
    const float   freq_scale  = *(const float *)((const int32_t *)dst->op_params + 6);

    GGML_ASSERT(mode == 0);  // gated by supports_op
    GGML_UNUSED(n_ctx_orig);

    // ggml shape: x->ne = [head_dim, n_heads, n_tokens, 1] (or similar 3D).
    // topsvllm wants query: [n_tokens, n_heads * head_size].
    // We treat the GCU memory as flat [n_tokens, n_heads * head_dim] using strides.
    const int64_t head_dim = x->ne[0];
    const int64_t n_heads  = x->ne[1];
    const int64_t n_tokens = x->ne[2];
    GGML_ASSERT(n_dims <= head_dim);
    GGML_ASSERT(pos->ne[0] == n_tokens);

    // 1) Build host cos/sin table of size [n_tokens + something, n_dims].
    //    Use max_pos = max(positions) + 1; conservatively read positions back
    //    once to determine the upper bound. This is a one-time host read per
    //    ROPE call; not on the critical path for correctness, only perf.
    std::vector<int32_t> pos_host(n_tokens);
    TOPS_CHECK(topsMemcpy(pos_host.data(), pos->data,
                          n_tokens * sizeof(int32_t), topsMemcpyDeviceToHost));
    int max_pos = 0;
    for (int32_t p : pos_host) max_pos = std::max(max_pos, (int) p);
    max_pos += 1;

    std::vector<float> cos_sin_host;
    gcu_build_rope_cos_sin_host(n_dims, max_pos, freq_base, freq_scale, cos_sin_host);

    // 2) Allocate device buffers and upload cos_sin + cast positions to I64.
    const size_t cs_bytes  = cos_sin_host.size() * sizeof(float);
    const size_t pos_bytes = n_tokens * sizeof(int64_t);
    void * cs_dev  = ctx->pool.alloc(cs_bytes);
    void * pos_dev = ctx->pool.alloc(pos_bytes);

    TOPS_CHECK(topsMemcpyAsync(cs_dev, cos_sin_host.data(), cs_bytes,
                               topsMemcpyHostToDevice, ctx->stream));

    // Cast I32 → I64 on host (cheap, n_tokens elements).
    std::vector<int64_t> pos_i64(n_tokens);
    for (int i = 0; i < n_tokens; i++) pos_i64[i] = (int64_t) pos_host[i];
    TOPS_CHECK(topsMemcpyAsync(pos_dev, pos_i64.data(), pos_bytes,
                               topsMemcpyHostToDevice, ctx->stream));

    // 3) topsvllmRotaryEmbedding query/key shape: [n_tokens, num_heads * head_size].
    //    For ggml's per-tensor ROPE we treat x as query; pass a small dummy as key.
    //    First, copy x → dst (since the op is in-place on query) — actually it's
    //    INOUT, so dst->data must equal x->data going in. ggml's ggml_rope creates
    //    a view of x: dst aliases x. So dst->data == x->data already.
    //    Confirm by assertion:
    GGML_ASSERT(dst->data == x->data);

    // 4) Allocate dummy key — minimum useful shape is [n_tokens, head_dim].
    const size_t dummy_key_bytes = (size_t) n_tokens * head_dim * ggml_type_size(x->type);
    void * dummy_key_dev = ctx->pool.alloc(dummy_key_bytes);
    TOPS_CHECK(topsMemsetAsync(dummy_key_dev, 0, dummy_key_bytes, ctx->stream));

    int64_t q_dims[2] = { n_tokens, n_heads * head_dim };
    int64_t q_strs[2] = { (int64_t)(x->nb[2] / ggml_type_size(x->type)), 1 };
    topsatenTensor q_tt(topsatenSize_t(q_dims, 2), topsatenSize_t(q_strs, 2),
                        ggml_to_topsaten_dtype(x->type), x->data);

    int64_t k_dims[2] = { n_tokens, head_dim };
    int64_t k_strs[2] = { head_dim, 1 };
    topsatenTensor k_tt(topsatenSize_t(k_dims, 2), topsatenSize_t(k_strs, 2),
                        ggml_to_topsaten_dtype(x->type), dummy_key_dev);

    int64_t pos_d[1] = { n_tokens };
    int64_t pos_s[1] = { 1 };
    topsatenTensor pos_tt(topsatenSize_t(pos_d, 1), topsatenSize_t(pos_s, 1),
                          TOPSATEN_DATA_I64, pos_dev);

    int64_t cs_d[2] = { max_pos, n_dims };
    int64_t cs_s[2] = { n_dims, 1 };
    topsatenTensor cs_tt(topsatenSize_t(cs_d, 2), topsatenSize_t(cs_s, 2),
                         TOPSATEN_DATA_FP32, cs_dev);

    TOPSATEN_CHECK(topsvllmRotaryEmbedding(q_tt, k_tt, pos_tt, cs_tt,
                                           (int) head_dim, /*is_neox=*/false,
                                           ctx->stream));

    TOPS_CHECK(topsStreamSynchronize(ctx->stream));
    ctx->pool.free(cs_dev,        cs_bytes);
    ctx->pool.free(pos_dev,       pos_bytes);
    ctx->pool.free(dummy_key_dev, dummy_key_bytes);
    return true;
}
```

```cpp
static bool gcu_op_mul_mat(ggml_backend_gcu_context * ctx, ggml_tensor * dst) {
    ggml_tensor * w = dst->src[0];
    ggml_tensor * x = dst->src[1];

    const ggml_type wt = w->type;
    const ggml_type xt = x->type;
    const ggml_type ot = dst->type;

    auto build_2d = [](const ggml_tensor * t, int64_t (& d)[2], int64_t (& s)[2]) {
        const size_t bpe = ggml_type_size(t->type);
        d[0] = t->ne[1];
        d[1] = t->ne[0];
        s[0] = (int64_t) (t->nb[1] / bpe);
        s[1] = (int64_t) (t->nb[0] / bpe);
    };

    // All-F32 fast path (existing behavior).
    if (wt == GGML_TYPE_F32 && xt == GGML_TYPE_F32 && ot == GGML_TYPE_F32) {
        int64_t lhs_d[2], lhs_s[2], rhs_d[2], rhs_s[2], out_d[2], out_s[2];
        build_2d(x,   lhs_d, lhs_s);
        build_2d(w,   rhs_d, rhs_s);
        build_2d(dst, out_d, out_s);

        topsatenTensor lhs(topsatenSize_t(lhs_d, 2), topsatenSize_t(lhs_s, 2),
                           ggml_to_topsaten_dtype(xt), x->data);
        topsatenTensor rhs(topsatenSize_t(rhs_d, 2), topsatenSize_t(rhs_s, 2),
                           ggml_to_topsaten_dtype(wt), w->data);
        topsatenTensor out(topsatenSize_t(out_d, 2), topsatenSize_t(out_s, 2),
                           ggml_to_topsaten_dtype(ot), dst->data);

        const int64_t M = dst->ne[0];
        const size_t  bias_bytes = (size_t) M * ggml_type_size(ot);
        void * bias_dev = gcu_get_zero_bias(ctx, bias_bytes);
        int64_t bias_d[1] = { M }; int64_t bias_s[1] = { 1 };
        topsatenTensor bias(topsatenSize_t(bias_d, 1), topsatenSize_t(bias_s, 1),
                            ggml_to_topsaten_dtype(ot), bias_dev);
        TOPSATEN_CHECK(topsatenLinear(out, lhs, rhs, bias, ctx->stream));
        return true;
    }

    // F16 weight path: input is F16 or F32, output is F16 or F32.
    // Strategy: cast input to F16 (if needed), matmul in F16, cast output to dst dtype.
    if (wt == GGML_TYPE_F16) {
        const int64_t M = dst->ne[0];
        const int64_t K = w->ne[0];
        const int64_t N = x->ne[1];

        // 1) Allocate F16 input scratch (+ optional cast).
        void * x_data = x->data;
        size_t x_cast_bytes = 0;
        void * x_cast_buf   = nullptr;
        if (xt == GGML_TYPE_F32) {
            x_cast_bytes = (size_t) N * K * sizeof(uint16_t);
            x_cast_buf   = ctx->pool.alloc(x_cast_bytes);

            int64_t xd[2] = { N, K };
            int64_t xs[2] = { K, 1 };
            topsatenTensor x_f32(topsatenSize_t(xd, 2), topsatenSize_t(xs, 2),
                                 TOPSATEN_DATA_FP32, x->data);
            topsatenTensor x_f16(topsatenSize_t(xd, 2), topsatenSize_t(xs, 2),
                                 TOPSATEN_DATA_FP16, x_cast_buf);
            topsatenDataType_t target = TOPSATEN_DATA_FP16;
            TOPSATEN_CHECK(topsatenTo(x_f16, x_f32, target, false, true,
                                      TOPSATEN_MEMORY_CONTIGUOUS, ctx->stream));
            x_data = x_cast_buf;
        }

        // 2) Allocate F16 output scratch.
        const size_t y_f16_bytes = (size_t) N * M * sizeof(uint16_t);
        void * y_f16 = ctx->pool.alloc(y_f16_bytes);

        // 3) F16 Linear with F16 zero bias.
        const size_t bias_bytes = (size_t) M * sizeof(uint16_t);
        void * bias_dev = gcu_get_zero_bias(ctx, bias_bytes);
        // gcu_get_zero_bias uses dst dtype-size sizing; if it's been called with
        // F32 sizes earlier it's already large enough. We rely on the bytes
        // being a multiple of FP16 size (true since FP16 < FP32).

        int64_t lhs_d[2] = { N, K }, lhs_s[2] = { K, 1 };
        int64_t rhs_d[2] = { M, K }, rhs_s[2] = { K, 1 };  // weight in F16 [M, K] with stride K (already contiguous)
        int64_t out_d[2] = { N, M }, out_s[2] = { M, 1 };
        int64_t bias_d[1] = { M };  int64_t bias_s[1] = { 1 };

        topsatenTensor lhs_f16(topsatenSize_t(lhs_d, 2), topsatenSize_t(lhs_s, 2),
                               TOPSATEN_DATA_FP16, x_data);
        topsatenTensor rhs_f16(topsatenSize_t(rhs_d, 2), topsatenSize_t(rhs_s, 2),
                               TOPSATEN_DATA_FP16, w->data);
        topsatenTensor out_f16(topsatenSize_t(out_d, 2), topsatenSize_t(out_s, 2),
                               TOPSATEN_DATA_FP16, y_f16);
        topsatenTensor bias_f16(topsatenSize_t(bias_d, 1), topsatenSize_t(bias_s, 1),
                                TOPSATEN_DATA_FP16, bias_dev);

        TOPSATEN_CHECK(topsatenLinear(out_f16, lhs_f16, rhs_f16, bias_f16, ctx->stream));

        // 4) Cast output F16 → dst dtype if needed.
        if (ot == GGML_TYPE_F32) {
            int64_t od[2] = { N, M }, os[2] = { M, 1 };
            topsatenTensor out_f32(topsatenSize_t(od, 2), topsatenSize_t(os, 2),
                                   TOPSATEN_DATA_FP32, dst->data);
            topsatenDataType_t target = TOPSATEN_DATA_FP32;
            TOPSATEN_CHECK(topsatenTo(out_f32, out_f16, target, false, true,
                                      TOPSATEN_MEMORY_CONTIGUOUS, ctx->stream));
        } else {
            // F16 output: copy y_f16 → dst.
            TOPS_CHECK(topsMemcpyAsync(dst->data, y_f16, y_f16_bytes,
                                       topsMemcpyDeviceToDevice, ctx->stream));
        }

        TOPS_CHECK(topsStreamSynchronize(ctx->stream));
        if (x_cast_buf) ctx->pool.free(x_cast_buf, x_cast_bytes);
        ctx->pool.free(y_f16, y_f16_bytes);
        return true;
    }

    GGML_ABORT("ggml-gcu MUL_MAT: unsupported dtype combo wt=%d xt=%d ot=%d",
               (int) wt, (int) xt, (int) ot);
}
```

```cpp
static int test_rope(ggml_backend_t gcu) {
    const int64_t head_dim = 64, n_heads = 8, n_tokens = 16;
    const int n_dims = (int) head_dim;
    const float freq_base = 10000.0f;
    auto buft = ggml_backend_get_default_buffer_type(gcu);
    ggml_init_params p = { ggml_tensor_overhead() * 32 + ggml_graph_overhead(), nullptr, true };
    ggml_context * ctx = ggml_init(p);

    ggml_tensor * a   = ggml_new_tensor_3d(ctx, GGML_TYPE_F32, head_dim, n_heads, n_tokens);
    ggml_tensor * pos = ggml_new_tensor_1d(ctx, GGML_TYPE_I32, n_tokens);
    ggml_tensor * c   = ggml_rope(ctx, a, pos, n_dims, GGML_ROPE_TYPE_NORMAL);

    ggml_backend_buffer_t buf = ggml_backend_alloc_ctx_tensors_from_buft(ctx, buft);
    if (!buf) { fprintf(stderr, "ROPE: alloc failed\n"); ggml_free(ctx); return 1; }

    const size_t n = head_dim * n_heads * n_tokens;
    std::vector<float>   ha(n), hc(n), expected(n);
    std::vector<int32_t> hp(n_tokens);
    fill_random_f32(ha.data(), n, 13);
    for (int64_t t = 0; t < n_tokens; t++) hp[t] = (int32_t) t;

    // CPU reference: ggml mode-0 RoPE.
    // x_rot[2i]   = x[2i]*cos(theta_i*p) - x[2i+1]*sin(theta_i*p)
    // x_rot[2i+1] = x[2i]*sin(theta_i*p) + x[2i+1]*cos(theta_i*p)
    for (int64_t t = 0; t < n_tokens; t++) {
        const int p = hp[t];
        for (int64_t h = 0; h < n_heads; h++) {
            const float * xrow = ha.data() + (t * n_heads + h) * head_dim;
            float       * yrow = expected.data() + (t * n_heads + h) * head_dim;
            const int half = n_dims / 2;
            for (int i = 0; i < half; i++) {
                const float theta = std::pow(freq_base, -2.0f * i / n_dims);
                const float angle = (float) p * theta;
                const float c1 = std::cos(angle), s1 = std::sin(angle);
                const float x0 = xrow[2 * i];
                const float x1 = xrow[2 * i + 1];
                yrow[2 * i]     = x0 * c1 - x1 * s1;
                yrow[2 * i + 1] = x0 * s1 + x1 * c1;
            }
            // For dims beyond n_dims, copy through.
            for (int i = n_dims; i < head_dim; i++) yrow[i] = xrow[i];
        }
    }

    ggml_backend_tensor_set(a, ha.data(), 0, n * sizeof(float));
    ggml_backend_tensor_set(pos, hp.data(), 0, n_tokens * sizeof(int32_t));
    ggml_cgraph * graph = ggml_new_graph(ctx); ggml_build_forward_expand(graph, c);
    if (ggml_backend_graph_compute(gcu, graph) != GGML_STATUS_SUCCESS) { fprintf(stderr, "ROPE: compute failed\n"); ggml_backend_buffer_free(buf); ggml_free(ctx); return 1; }
    ggml_backend_tensor_get(c, hc.data(), 0, n * sizeof(float));

    int bad = 0;
    for (size_t i = 0; i < n; i++) if (!close_enough(hc[i], expected[i], 1e-3f, 1e-3f)) { if (bad++ < 5) fprintf(stderr, "ROPE mismatch %zu got=%f want=%f\n", i, hc[i], expected[i]); }
    ggml_backend_buffer_free(buf); ggml_free(ctx);
    if (bad) return 1;
    printf("ROPE: ok (%zu elements)\n", n); return 0;
}
```

```markdown
### Operator coverage (MVP-2)

The backend implements the following ops on GCU:

- Element-wise: ADD, MUL, SUB (no), DIV (no), SCALE (bias=0 only)
- Activations: SILU
- Normalization: RMS_NORM
- Position encoding: ROPE (mode 0 only — no NEOX, no YARN, no MROPE)

- Linear: MUL_MAT (F32×F32→F32 fast path, F16-weight×{F16,F32}→{F16,F32} via cast)

- Memory: CPY/DUP/CONT (same dtype contiguous + F32↔F16 conversion), view ops (RESHAPE/VIEW/PERMUTE/TRANSPOSE)

### Known limitations (MVP-2)

- Quantized weights (Q4_K, Q8_0, etc.) stay on CPU. Models load fine but compute is CPU-bound.
- KV cache stays on CPU by design. SET_ROWS/GET_ROWS for cache-shaped tensors are explicitly refused; the scheduler routes them to CPU. This caps throughput on real models but unblocks all attention math on GCU.
- BF16 not supported.
- Only ROPE mode 0 is implemented; YARN/NEOX/MROPE go to CPU.
- SOFT_MAX with `max_bias != 0` (alibi) goes to CPU.
- Single device, single stream, no pinned host memory. Multi-device, async overlap, native quantized matmul are MVP-3 work.
```
